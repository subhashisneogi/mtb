)

        "purchase_order__version": "PO Version",
        "purchase_order__delivery_site__site_name": "Delivery Site",

class PurchaseOrderItemsSReportAPIView(APIView):

    def get(self, request):
        check_editor_permissions(request, "po")

        all = request.query_params.get("all", None)
        export = request.query_params.get("export", None)
        order_by = request.query_params.get("order_by", "-id")

        search = {"purchase_order__is_deleted": False}

        temp = check_store_site(request)

        if "site__in" in temp:
            search["purchase_order__site__in"] = temp["site__in"]

        if "store__in" in temp:
            search["purchase_order__store__in"] = temp["store__in"]

        search = custom_filters(request, search, [])

        grn_base_qs = GRNItems.cmobjects.filter(
            grn__is_deleted=False,
            purchase_order_item_id=OuterRef("pk")
        )

        grn_latest_qs = grn_base_qs.order_by("-grn__created_at", "-id")

        po_max_stage_qs = MultiStageSettingStages.cmobjects.filter(
            multi_stage_setting__is_deleted=False,
            multi_stage_setting_id=OuterRef("purchase_order__ref_stage_type_id")
        ).order_by("-stage").values("stage")[:1]

        data = PurchaseOrderItems.cmobjects.annotate(

            grn_quantity=Coalesce(
                Subquery(
                    grn_base_qs.values("purchase_order_item")
                    .annotate(total_quantity=Sum("quantity"))
                    .values("total_quantity")[:1],
                    output_field=FloatField()
                ),
                Value(0.0),
                output_field=FloatField()
            ),

            cst__request_code=Subquery(
                CSTItems.cmobjects.filter(
                    procurement_purchase_order_item_cst_item__id=OuterRef("pk"),
                    cst__latest=True,
                )
                .exclude(cst__status__in=["rejected", "cancelled"])
                .values("procurement_purchase_order_item_cst_item")
                .annotate(
                    value=GroupConcat(
                        Concat(
                            "cst__request_code",
                            Value(" V:"),
                            "cst__version",
                            output_field=CharField()
                        ),
                        distinct=True
                    )
                )
                .values("value")[:1],
                output_field=CharField()
            ),

            requisition_remarks=Subquery(
                MaterialRequestMasterItemsRemarks.cmobjects.filter(
                    material_request_item_id=OuterRef("material_request_item_id"),
                    remarks_type="general",
                )
                .order_by("-created_at")
                .values("remarks")[:1],
                output_field=CharField()
            ),

            purchase_remarks=Subquery(
                MaterialRequestMasterItemsRemarks.cmobjects.filter(
                    material_request_item_id=OuterRef("material_request_item_id"),
                    remarks_type="purchase",
                )
                .order_by("-created_at")
                .values("remarks")[:1],
                output_field=CharField()
            ),

            purchase_order_created_by_name=Concat(
                "purchase_order__created_by__first_name",
                Value(" "),
                "purchase_order__created_by__last_name",
                output_field=CharField()
            ),

            po_multi_stage_max_stage=Subquery(
                po_max_stage_qs,
                output_field=IntegerField()
            ),

            purchase_order_final_approved_by_date=Subquery(
                PurchaseOrderStages.cmobjects.filter(
                    purchase_order_id=OuterRef("purchase_order_id"),
                    status="approved",
                    stage=OuterRef("po_multi_stage_max_stage"),
                )
                .order_by("-created_at")
                .values("approved_by_date")[:1]
            ),

            purchase_order_final_approved_by=Subquery(
                PurchaseOrderStages.cmobjects.filter(
                    purchase_order_id=OuterRef("purchase_order_id"),
                    status="approved",
                    stage=OuterRef("po_multi_stage_max_stage"),
                )
                .order_by("-created_at")
                .annotate(
                    full_name=Concat(
                        "approved_by__first_name",
                        Value(" "),
                        "approved_by__last_name",
                        output_field=CharField()
                    )
                )
                .values("full_name")[:1],
                output_field=CharField()
            ),

            grn_reference=Subquery(grn_latest_qs.values("grn__reference")[:1]),
            grn_bill_date=Subquery(grn_latest_qs.values("grn__bill_date")[:1]),
            grn_sap_code=Subquery(grn_latest_qs.values("grn__sap_code")[:1]),
            grn_total_amount=Subquery(grn_latest_qs.values("grn__total_amount")[:1]),
            grn_item_po_item=Subquery(grn_latest_qs.values("po_item")[:1]),
            grn_created_date=Subquery(grn_latest_qs.values("grn__created_at")[:1]),
            grn_batch=Subquery(grn_latest_qs.values("grn__batch")[:1]),
            grn_date=Subquery(grn_latest_qs.values("grn__date")[:1]),

            purchase_group_description=F(
                "purchase_order__purchase_group_for_sap__description"
            ),

        ).filter(*search).values(

            "purchase_order__vendor__vendor_code",
            "purchase_order__vendor__vendor_name",
            "sap_item_no",
            "purchase_order__vendor__gstin_status_category",
            "purchase_order__vendor__gstin_status",
            "purchase_order__vendor__gst_class",
            "purchase_order__vendor__region",
            "purchase_order__vendor__vendor_classif_text",
            "purchase_order__vendor__vendor_address",
            "purchase_order__vendor__company_code",

            "purchase_order__sap_code",
            "purchase_order_created_by_name",
            "purchase_order_final_approved_by_date",
            "purchase_order_final_approved_by",
            "purchase_order__date",
            "purchase_order__request_code",
            "purchase_order__version",
            "purchase_order__delivery_site__site_name",
            "purchase_order__billing_site__site_name",
            "purchase_order__tax_type",
            "purchase_order__final_release_date_from_sap",
            "purchase_order__total_amount",

            "item__sap_code",
            "item__material_code",
            "item__material_type__name",
            "item__material_name",

            "material_request_item__material_request__request_code",

            "uom__formal_name",
            "uom__symbol",

            "cst__request_code",

            "item_amount",
            "total_amount",
            "taxable_amount",
            "quantity",
            "grn_quantity",
            "sgst_amount",
            "cgst_amount",
            "igst_amount",

            "grn_reference",
            "grn_bill_date",
            "grn_sap_code",
            "grn_total_amount",
            "grn_item_po_item",
            "grn_created_date",
            "grn_batch",
            "grn_date",

            "requisition_remarks",
            "purchase_remarks",
            "purchase_order__store__sap_plant_id",
            "purchase_group_description",

        ).order_by(*str(order_by).split(","))

        if export == "csv":
            return export_purchase_order_items_csv(data)

        if all == "true":
            return Response({
                "results": list(data)
            })

        page_size = int(request.query_params.get("page_size", settings.MIN_PAGE_SIZE))
        paginator = Paginator(data, page_size)

        page_number = request.query_params.get("page", 1)
        page = paginator.get_page(page_number)

        return Response({
            "count": paginator.count,
            "next": page.next_page_number() if page.has_next() else None,
            "previous": page.previous_page_number() if page.has_previous() else None,
            "results": {
                "Data": list(page.object_list),
            },
        })
